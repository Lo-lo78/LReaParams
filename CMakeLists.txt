cmake_minimum_required(VERSION 3.24)
project(LReaParams VERSION 0.0.3 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "LReaParams requires Windows x64 and the MSVC ABI")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(lrp_profiles STATIC src/Profile.cpp src/Storage.cpp)
target_include_directories(lrp_profiles PUBLIC include)
target_compile_definitions(lrp_profiles PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(lrp_profiles PUBLIC bcrypt)
add_library(reaper_lreaparams SHARED src/Extension.cpp src/InputRouter.cpp src/FxNavigator.cpp src/TrackNavigator.cpp src/ReaperApi.cpp)
target_include_directories(reaper_lreaparams PRIVATE include third_party/reaper-sdk)
target_link_libraries(reaper_lreaparams PRIVATE lrp_profiles user32 shell32)
set_target_properties(reaper_lreaparams PROPERTIES PREFIX "" OUTPUT_NAME "reaper_lreaparams")
foreach(target lrp_profiles reaper_lreaparams)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
endforeach()
enable_testing()
add_executable(profile_tests Testing/ProfileTests.cpp)
target_link_libraries(profile_tests PRIVATE lrp_profiles)
target_compile_options(profile_tests PRIVATE /W4 /permissive- /utf-8 /EHsc)
add_test(NAME profiles COMMAND profile_tests "${CMAKE_CURRENT_SOURCE_DIR}/Testing/fixtures")
